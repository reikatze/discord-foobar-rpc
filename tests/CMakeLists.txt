cmake_minimum_required(VERSION 3.20)
project(discord_foobar_rpc_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.11 CONFIG REQUIRED)
if(WIN32)
  find_package(CURL CONFIG REQUIRED)
else()
  find_package(CURL REQUIRED)
endif()
find_package(Threads REQUIRED)
set(MUSIC_SANITIZER "" CACHE STRING "Optional address,undefined or thread sanitizer")
if(MUSIC_SANITIZER AND NOT MSVC)
  add_compile_options("-fsanitize=${MUSIC_SANITIZER}" -fno-omit-frame-pointer -g)
  add_link_options("-fsanitize=${MUSIC_SANITIZER}")
endif()
add_library(music_core STATIC ../modules/model.cpp ../modules/presence.cpp ../modules/cache.cpp
  ../modules/http.cpp ../modules/discord.cpp ../modules/musicbrainz.cpp ../modules/service.cpp ../modules/settings_policy.cpp)
if(WIN32)
  target_sources(music_core PRIVATE ../platform/discord_windows.cpp)
  target_compile_definitions(music_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN CURL_STATICLIB)
else()
  target_sources(music_core PRIVATE ../platform/discord_macos.cpp)
  if(NOT APPLE)
    target_compile_definitions(music_core PRIVATE MUSIC_TRACKER_TEST_POSIX)
  endif()
endif()
target_link_libraries(music_core PUBLIC nlohmann_json::nlohmann_json CURL::libcurl Threads::Threads)
if(MSVC)
  target_compile_options(music_core PRIVATE /W4)
else()
  target_compile_options(music_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
foreach(name core discord musicbrainz settings cache service native_ipc)
  add_executable(${name}_tests ${name}_tests.cpp)
  target_link_libraries(${name}_tests PRIVATE music_core)
  add_test(NAME ${name} COMMAND ${name}_tests)
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
set_tests_properties(native_ipc PROPERTIES LABELS "native;integration")
set_tests_properties(service PROPERTIES LABELS "coordination;threading")
